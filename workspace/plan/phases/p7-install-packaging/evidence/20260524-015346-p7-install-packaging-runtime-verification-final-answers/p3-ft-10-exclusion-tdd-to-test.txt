파일 쓰기를 시도했지만 현재 세션이 `read-only sandbox`라서 실제 생성은 차단됐습니다. 작업 디렉터리도 비어 있고 git 저장소가 아니어서 기존 모델/앱에 맞춘 factory는 만들 수 없었습니다.

적용하려던 최소 구조는 아래입니다.

```diff
*** Add File: tests/__init__.py
+
*** Add File: tests/conftest.py
+from __future__ import annotations
+
+from collections.abc import Callable
+from typing import Any
+from unittest.mock import AsyncMock, Mock, seal
+
+import pytest
+from pytest_mock import MockerFixture
+
+
+@pytest.fixture
+def mock_external_client(mocker: MockerFixture) -> Mock:
+    return mocker.Mock(name="external_client")
+
+
+@pytest.fixture
+def mock_async_external_client(mocker: MockerFixture) -> AsyncMock:
+    return mocker.AsyncMock(name="async_external_client")
+
+
+@pytest.fixture
+def mock_response_factory(mocker: MockerFixture) -> Callable[..., Mock]:
+    def make_mock_response(
+        *,
+        status_code: int = 200,
+        json_data: Any | None = None,
+        text: str = "",
+    ) -> Mock:
+        response = mocker.Mock(name="response")
+        response.status_code = status_code
+        response.text = text
+        response.json.return_value = json_data if json_data is not None else {}
+        seal(response)
+        return response
+
+    return make_mock_response
*** Add File: tests/factories/__init__.py
+from tests.factories.base import BaseDictFactory
+
+__all__ = ["BaseDictFactory"]
*** Add File: tests/factories/base.py
+from __future__ import annotations
+
+import factory
+
+
+class BaseDictFactory(factory.Factory):
+    class Meta:
+        model = dict
+
+    id = factory.Sequence(lambda n: n + 1)
```

검증은 파일 생성이 차단되어 실행하지 못했습니다.
Serena: 사용할 수 있는 Serena 도구가 없고 현재 디렉터리가 빈 비-git 경로라서 생략했으며, `pwd`, `rg --files`, `find`로 구조를 확인했습니다.
