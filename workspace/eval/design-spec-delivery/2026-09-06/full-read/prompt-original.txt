class NoticeSpeechComposer:
    """방 안 안내 발화 프롬프트(D22 · §4.4 «방 안 = 점쟁이 말투 LLM» · 무상태) — 잡담 R4 와 별개 composer 다.

    (system 안내 지시문, system persona 블록, user 상황 요약) 셋만 만든다 — 안내는 한두 문장이라 대본 문맥(R2)을 넣지 않는다.
    지시문·라벨은 ko 코드 상수이고 LLM 에게 «`language.code` 언어로 답하라» 만 시킨다(언어 축 없음 — `settings.LANGUAGES` 가
    늘면 표에 언어 축을 더한다). 경계 표식은 `SmallTalkPromptComposer` 와 한 벌이다(D47-1).
    """

    SITUATION_LINES: ClassVar[Mapping[SpeechSituation, str]] = MappingProxyType(
        {
            SpeechSituation.OTHER_CHARACTER_GUIDANCE: "손님이 원하는 운세 종류는 네가 볼 수 없고 다른 점쟁이가 볼 수 있다고 안내한다",
            SpeechSituation.NO_CHARACTER: "지금은 그 운세를 볼 수 있는 점쟁이가 없다고 안내한다",
            SpeechSituation.PROFILE_INPUT_REQUEST: "운세를 보려면 손님의 내 정보 중 {subject}가 필요하다고 안내한다",
            SpeechSituation.REQUEST_CANCELLED: "손님이 이번 요청을 접었음을 확인하고 다시 말하면 이어서 돕겠다고 한다",
        }
    )
    INPUT_LABELS: ClassVar[Mapping[RequiredInput, str]] = MappingProxyType(
        {
            RequiredInput.BIRTH_DATETIME: "생년월일시",
            RequiredInput.BIRTH_PLACE: "출생지",
            RequiredInput.SEX: "성별",
        }
    )

    @classmethod
    def compose(
        cls,
        persona: str,
        language: DisplayLanguage,
        situation: SpeechSituation,
        subject: str | None,
        customer_message: str | None,
    ) -> tuple[SpeechMessage, ...]:
        """표 밖 상황은 `InvalidSpeechSituation`(§4.4 폴백이 접는다) · `subject` 는 상황 문장의 `{subject}` 자리에만 쓰인다."""
        line: str | None = cls.SITUATION_LINES.get(situation)
        if line is None:
            raise InvalidSpeechSituation(f"안내 발화 표에 없는 상황입니다: {situation.value}")
        situation_line: str = line.format(subject="" if subject is None else subject)
        instruction: SpeechMessage = SpeechMessage(
            role="system", content=_INSTRUCTION_TEMPLATE.format(language=language.code, situation=situation_line)
        )
        persona_block: SpeechMessage = SpeechMessage(
            role="system",
            content=f"{SmallTalkPromptComposer.PERSONA_OPEN}\n{persona}\n{SmallTalkPromptComposer.PERSONA_CLOSE}",
        )
        summary: str = (
            _CARD_ANSWER_TURN_MESSAGE
            if customer_message is None
            else _CUSTOMER_MESSAGE_TEMPLATE.format(customer_message=customer_message)
        )
        return (instruction, persona_block, SpeechMessage(role="user", content=summary))
