            SpeechSituation.REQUEST_CANCELLED: "손님이 이번 요청을 접었음을 확인하고 다시 말하면 이어서 돕겠다고 한다",

            RequiredInput.SEX: "성별",
        }
    )

    def compose(

            if customer_message is None
            else _CUSTOMER_MESSAGE_TEMPLATE.format(customer_message=customer_message)
        )
