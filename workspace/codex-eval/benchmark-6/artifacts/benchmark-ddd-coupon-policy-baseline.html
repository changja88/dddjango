+from __future__ import annotations
+
+from abc import ABC, abstractmethod
+from dataclasses import dataclass, field, replace
+from datetime import datetime
+from enum import Enum
+
+
+class CouponDomainError(Exception):
+    pass
+
+
+class CouponNotApplicableError(CouponDomainError):
+    pass
+
+
+class CouponAlreadyRedeemedError(CouponDomainError):
+    pass
+
+
+class CouponCombinationNotAllowedError(CouponDomainError):
+    pass
+
+
+@dataclass(frozen=True, slots=True)
+class Money:
+    amount: int
+    currency: str = "KRW"
+
+    def __post_init__(self) -> None:
+        if self.amount < 0:
+            raise CouponDomainError("금액은 0 이상이어야 합니다")
+        if not self.currency:
+            raise CouponDomainError("통화는 필수입니다")
+
+    def subtract(self, other: Money) -> Money:
+        self._ensure_same_currency(other)
+        return replace(self, amount=max(0, self.amount - other.amount))
+
+    def min(self, other: Money) -> Money:
+        self._ensure_same_currency(other)
+        return self if self.amount <= other.amount else other
+
+    def _ensure_same_currency(self, other: Money) -> None:
+        if self.currency != other.currency:
+            raise CouponDomainError("통화가 다릅니다")
+
+
+@dataclass(frozen=True, slots=True)
+class OrderSnapshot:
+    order_id: str
+    customer_id: str
+    total: Money
+
+    def __post_init__(self) -> None:
+        if not self.order_id:
+            raise CouponDomainError("주문 ID는 필수입니다")
+        if not self.customer_id:
+            raise CouponDomainError("고객 ID는 필수입니다")
+
+
+class DiscountType(Enum):
+    FIXED_AMOUNT = "fixed_amount"
+    PERCENT_RATE = "percent_rate"
+
+
+@dataclass(frozen=True, slots=True)
+class DiscountPolicy:
+    discount_type: DiscountType
+    fixed_amount: Money | None = None
+    percent_rate: int | None = None
+
+    def __post_init__(self) -> None:
+        if self.discount_type is DiscountType.FIXED_AMOUNT:
+            if self.fixed_amount is None or self.fixed_amount.amount <= 0:
+                raise CouponDomainError("정액 할인 금액은 0보다 커야 합니다")
+            if self.percent_rate is not None:
+                raise CouponDomainError("정액 할인에는 할인율을 함께 둘 수 없습니다")
+        if self.discount_type is DiscountType.PERCENT_RATE:
+            if self.percent_rate is None or not 1 <= self.percent_rate <= 100:
+                raise CouponDomainError("정률 할인율은 1 이상 100 이하입니다")
+            if self.fixed_amount is not None:
+                raise CouponDomainError("정률 할인에는 정액 금액을 함께 둘 수 없습니다")
+
+    def calculate_discount(self, order_total: Money) -> Money:
+        if self.discount_type is DiscountType.FIXED_AMOUNT:
+            assert self.fixed_amount is not None
+            return self.fixed_amount.min(order_total)
+
+        assert self.percent_rate is not None
+        return Money(order_total.amount * self.percent_rate // 100, order_total.currency)
+
+
+@dataclass(frozen=True, slots=True)
+class CouponPolicy:
+    minimum_order_amount: Money
+    combinable: bool
+    once_per_customer: bool = True
+
+    def assert_satisfied_by(self, order: OrderSnapshot) -> None:
+        self.minimum_order_amount._ensure_same_currency(order.total)
+        if order.total.amount < self.minimum_order_amount.amount:
+            raise CouponNotApplicableError("최소 주문 금액을 충족하지 못했습니다")
+
+
+@dataclass(frozen=True, slots=True)
+class CouponAppliedEvent:
+    coupon_id: str
+    coupon_code: str
+    order_id: str
+    customer_id: str
+    discount_amount: Money
+    occurred_at: datetime
+
+
+@dataclass
+class Coupon:
+    """Coupon 애그리거트 루트.
+
+    불변식:
+    - 쿠폰은 자신의 할인 방식과 적용 조건을 스스로 검증한다.
+    - 최소 주문 금액 미달 주문에는 적용할 수 없다.
+    - once_per_customer=True이면 같은 고객은 같은 쿠폰을 다시 사용할 수 없다.
+    - combinable=False이면 다른 쿠폰과 함께 적용할 수 없다.
+    """
+
+    id: str
+    code: str
+    discount_policy: DiscountPolicy
+    coupon_policy: CouponPolicy
+    _events: list[CouponAppliedEvent] = field(default_factory=list)
+
+    def apply_to(
+        self,
+        order: OrderSnapshot,
+        *,
+        already_redeemed_by_customer: bool,
+        other_coupon_count: int,
+        now: datetime,
+    ) -> Money:
+        self.coupon_policy.assert_satisfied_by(order)
+
+        if self.coupon_policy.once_per_customer and already_redeemed_by_customer:
+            raise CouponAlreadyRedeemedError("이미 사용한 쿠폰입니다")
+
+        if not self.coupon_policy.combinable and other_coupon_count > 0:
+            raise CouponCombinationNotAllowedError("다른 쿠폰과 함께 사용할 수 없습니다")
+
+        discount_amount = self.discount_policy.calculate_discount(order.total)
+        self._record_event(
+            CouponAppliedEvent(
+                coupon_id=self.id,
+                coupon_code=self.code,
+                order_id=order.order_id,
+                customer_id=order.customer_id,
+                discount_amount=discount_amount,
+                occurred_at=now,
+            )
+        )
+        return discount_amount
+
+    def _record_event(self, event: CouponAppliedEvent) -> None:
+        self._events.append(event)
+
+    def collect_events(self) -> list[CouponAppliedEvent]:
+        events = list(self._events)
+        self._events.clear()
+        return events
+
+
+class CouponRepository(ABC):
+    @abstractmethod
+    def find_by_code(self, code: str) -> Coupon | None:
+        ...
+
+    @abstractmethod
+    def save(self, coupon: Coupon) -> None:
+        ...
+
+
+class CouponRedemptionRepository(ABC):
+    @abstractmethod
+    def has_redeemed(self, coupon_id: str, customer_id: str) -> bool:
+        ...
+
+    @abstractmethod
+    def save_redemption(self, event: CouponAppliedEvent) -> None:
+        ...

+from __future__ import annotations
+
+from dataclasses import dataclass

+from coupons.domain.model import (

+)

+@dataclass(frozen=True, slots=True)
+class ApplyCouponCommand:
+    coupon_code: str
+    order_id: str
+    customer_id: str
+    order_total: Money
+    already_applied_coupon_count: int = 0
+
+
+@dataclass(frozen=True, slots=True)
+class ApplyCouponResult:
+    coupon_code: str
+    discount_amount: Money
+    payable_amount: Money
+
+
+class CouponApplicationService:
+    """쿠폰 적용 유스케이스 조율자.
+
+    조회, 중복 사용 이력 확인, 저장만 조율하고 쿠폰 정책 판단은 Coupon 애그리거트에 위임한다.
+    """
+
+    def __init__(
+        self,
+        coupon_repository: CouponRepository,
+        redemption_repository: CouponRedemptionRepository,
+    ) -> None:
+        self._coupon_repository = coupon_repository
+        self._redemption_repository = redemption_repository
+
+    def apply_coupon(self, command: ApplyCouponCommand) -> ApplyCouponResult:
+        coupon = self._coupon_repository.find_by_code(command.coupon_code)
+        if coupon is None:
+            raise CouponNotApplicableError("쿠폰을 찾을 수 없습니다")
+
+        order = OrderSnapshot(
+            order_id=command.order_id,
+            customer_id=command.customer_id,
+            total=command.order_total,
+        )
+        already_redeemed = self._redemption_repository.has_redeemed(
+            coupon.id,
+            command.customer_id,
+        )
+
+        discount_amount = coupon.apply_to(
+            order,
+            already_redeemed_by_customer=already_redeemed,
+            other_coupon_count=command.already_applied_coupon_count,
+            now=datetime.now(),
+        )
+
+        for event in coupon.collect_events():
+            self._redemption_repository.save_redemption(event)
+        self._coupon_repository.save(coupon)
+
+        return ApplyCouponResult(
+            coupon_code=coupon.code,
+            discount_amount=discount_amount,
+            payable_amount=command.order_total.subtract(discount_amount),
+        )
